cmake_minimum_required(VERSION 3.16)
project(SwapchainAcquireTests LANGUAGES CXX)
enable_testing()
set(TEMPEST_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(TEMPEST_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
add_subdirectory(../../Engine tempest)
add_executable(swapchain-acquire-tests main.cpp)
target_compile_features(swapchain-acquire-tests PRIVATE cxx_std_20)
target_include_directories(swapchain-acquire-tests PRIVATE ../../Engine ../../Engine/include)
target_link_libraries(swapchain-acquire-tests PRIVATE Tempest)
target_link_options(swapchain-acquire-tests PRIVATE
  -Wl,--wrap=vkAcquireNextImageKHR -Wl,--wrap=vkWaitForFences -Wl,--wrap=vkResetFences)
add_test(NAME swapchain-acquire-timeout-retry COMMAND swapchain-acquire-tests)
set_tests_properties(swapchain-acquire-timeout-retry PROPERTIES TIMEOUT 30)
